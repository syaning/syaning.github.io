**Approach:** Reject negatives; reverse digits with mod/div and compare to the original.

**Complexity:** O(log x) time, O(1) space

::: code-group

```c++
class Solution {
public:
    bool isPalindrome(int x) {
        if (x < 0) {
            return false;
        }
        int copy = x;
        int reversed = 0;

        while (copy > 0) {
            reversed = reversed * 10 + copy % 10;
            copy = copy / 10;
        }

        return reversed == x;
    }
};
```

```go
func isPalindrome(x int) bool {
    if x < 0 {
        return false
    }

    copyed, reversed := x, 0

    for copyed > 0 {
        reversed = reversed*10 + copyed%10
        copyed = copyed / 10
    }

    return x == reversed
}
```

```js
/**

var isPalindrome = function(x) {
  if (x < 0) {
    return false
  }

  var copy = x
  var reverse = 0

  while (copy) {
    reverse = reverse * 10 + copy % 10
    copy = copy / 10 >> 0
  }

  return x === reverse
}
```

```py
class Solution(object):

    def isPalindrome(self, x):
        """
        :type x: int
        :rtype: bool
        """
        if x < 0:
            return False
        copy, reverse = x, 0
        while copy:
            reverse = reverse * 10 + copy % 10
            copy = copy // 10
        return x == reverse
```
