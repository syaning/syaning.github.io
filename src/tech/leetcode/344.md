```js [JavaScript]
/**
 * @param {string} s
 * @return {string}
 */
var reverseString = function(s) {
  return s.split('').reverse().join('')
}
```

```py [Python]
class Solution:

    def reverseString(self, s):
        """
        :type s: str
        :rtype: str
        """
        return s[::-1]
```
