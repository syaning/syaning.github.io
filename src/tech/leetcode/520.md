```js [JavaScript]
/**
 * @param {string} word
 * @return {boolean}
 */
var detectCapitalUse = function(word) {
  return word.toUpperCase() === word || word.slice(1).toLowerCase() === word.slice(1)
}
```

```py [Python]
class Solution:

    def detectCapitalUse(self, word):
        """
        :type word: str
        :rtype: bool
        """
        return word.upper() == word or word[1:].lower() == word[1:]
```
