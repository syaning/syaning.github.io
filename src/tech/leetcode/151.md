```js [JavaScript]
/**
 * @param {string} str
 * @returns {string}
 */
var reverseWords = function(str) {
  return str.trim().split(/\s+/).reverse().join(' ')
}
```

```py [Python]
class Solution(object):

    def reverseWords(self, s):
        """
        :type s: str
        :rtype: str
        """
        return ' '.join(reversed(s.strip().split()))
```
