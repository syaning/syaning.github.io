```js
/**
 * @param {string} str
 * @return {string}
 */
var toLowerCase = function(str) {
  return str.toLowerCase()
}
```

```py
class Solution:

    def toLowerCase(self, str):
        """
        :type str: str
        :rtype: str
        """
        return str.lower()
```
